Aturan Firestore (Pencarian Aman per-NIM) 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /settings/{doc} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /kelulusan/{nim} {
      allow get: if true;
      allow list, write: if request.auth != null;
    }
  }
}
